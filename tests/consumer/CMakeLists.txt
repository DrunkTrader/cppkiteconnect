cmake_minimum_required(VERSION 3.18)
project(kiteppIndependentConsumer LANGUAGES CXX)
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
get_filename_component(KITEPP_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(KITEPP_CONSUMPTION "subdirectory" CACHE STRING "headers, subdirectory, or installed")
set_property(CACHE KITEPP_CONSUMPTION PROPERTY STRINGS headers subdirectory installed)
option(KITEPP_CONSUMER_TICKER "Consume ticker and umbrella as well as REST/models" ON)
if(KITEPP_CONSUMPTION STREQUAL "headers")
    find_package(OpenSSL REQUIRED)
    find_package(Threads REQUIRED)
    add_library(consumer_headers INTERFACE)
    target_compile_features(consumer_headers INTERFACE cxx_std_20)
    target_include_directories(consumer_headers INTERFACE "${KITEPP_SOURCE}/include")
    target_link_libraries(consumer_headers INTERFACE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
    if(KITEPP_CONSUMER_TICKER)
        find_package(Boost 1.83 CONFIG REQUIRED)
        target_link_libraries(consumer_headers INTERFACE Boost::headers)
    endif()
elseif(KITEPP_CONSUMPTION STREQUAL "subdirectory")
    set(KITEPP_ENABLE_TICKER ${KITEPP_CONSUMER_TICKER} CACHE BOOL "" FORCE)
    set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${KITEPP_SOURCE}" sdk)
elseif(KITEPP_CONSUMPTION STREQUAL "installed")
    if(KITEPP_CONSUMER_TICKER)
        find_package(kitepp 2.2 CONFIG REQUIRED COMPONENTS models rest ticker kitepp)
    else()
        find_package(kitepp 2.2 CONFIG REQUIRED COMPONENTS models rest)
        if(TARGET Boost::headers)
            message(FATAL_ERROR "REST/model-only discovery unexpectedly imported Boost")
        endif()
    endif()
else()
    message(FATAL_ERROR "Unknown KITEPP_CONSUMPTION: ${KITEPP_CONSUMPTION}")
endif()

function(consume target component)
    if(KITEPP_CONSUMPTION STREQUAL "headers")
        target_link_libraries(${target} PRIVATE consumer_headers)
    else()
        target_link_libraries(${target} PRIVATE kitepp::${component})
    endif()
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS NO)
    add_test(NAME ${target} COMMAND ${target})
endfunction()

enable_testing()
set(components rest standalone)
if(KITEPP_CONSUMER_TICKER)
    list(APPEND components ticker umbrella)
endif()
foreach(component IN LISTS components)
    add_executable(consumer_${component} "../compile/${component}.cpp")
    if(component STREQUAL "standalone")
        consume(consumer_${component} models)
    elseif(component STREQUAL "umbrella")
        consume(consumer_${component} kitepp)
    else()
        consume(consumer_${component} ${component})
    endif()
endforeach()
add_executable(consumer_two_tu "../compile/two_tu_main.cpp"
    "../compile/two_tu_a.cpp" "../compile/two_tu_b.cpp")
consume(consumer_two_tu rest)
if(KITEPP_CONSUMER_TICKER)
    add_executable(consumer_ticker_two_tu "../compile/ticker_two_tu_main.cpp"
        "../compile/ticker_two_tu_a.cpp" "../compile/ticker_two_tu_b.cpp")
    consume(consumer_ticker_two_tu kitepp)
endif()
