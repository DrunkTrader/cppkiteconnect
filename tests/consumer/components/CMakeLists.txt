cmake_minimum_required(VERSION 3.18)
project(kiteppComponentChecks LANGUAGES CXX)
set(KITEPP_REQUIRED_COMPONENT rest CACHE STRING "Component that must be present")
option(KITEPP_EXPECT_TICKER "Expect optional ticker to be available" ON)
if(KITEPP_REQUIRED_COMPONENT STREQUAL "ticker")
    find_package(kitepp 2.2 CONFIG REQUIRED COMPONENTS ticker)
else()
    find_package(kitepp 2.2 CONFIG REQUIRED COMPONENTS ${KITEPP_REQUIRED_COMPONENT}
        OPTIONAL_COMPONENTS ticker)
endif()
if(NOT TARGET kitepp::rest OR NOT TARGET kitepp::models)
    message(FATAL_ERROR "Core component targets missing")
endif()
if(KITEPP_EXPECT_TICKER AND NOT kitepp_ticker_FOUND)
    message(FATAL_ERROR "Expected optional ticker")
elseif(NOT KITEPP_EXPECT_TICKER AND kitepp_ticker_FOUND)
    message(FATAL_ERROR "Unexpected optional ticker")
endif()
